1//! Retention for conversation transcripts.
2//!
3//! Transcripts past the retention window are removed from `~/.claude/projects`
4//! and the sync repository together: removing them on one side only means the
5//! next sync restores them from the other.
6//!
7//! The window is never shorter than `cleanupPeriodDays`, which is when Claude
8//! Code deletes the transcript anyway, nor shorter than six months.
9//!
10//! The desktop app's session entries that point at a purged transcript go
11//! with it, each leaving the tombstone the app itself writes for a deleted
12//! session (see `crate::artifacts::desktop_sessions`), so no machine keeps
13//! listing the session or re-adopts the transcript.
14

24/// Shortest retention window applied by default: six months.
25pub const MINIMUM_RETENTION_DAYS: u32 = 180;
26
27/// Claude Code's documented default for `cleanupPeriodDays`.
28pub const CLAUDE_CODE_DEFAULT_RETENTION_DAYS: u32 = 30;

30/// One session that has aged out, with every file belonging to it.
31#[derive(Debug, Clone)]
32pub struct PurgeTarget {
33    pub session_id: String,
34    /// Timestamp of the last message, which the age is measured from.
35    pub last_activity: DateTime<Utc>,
36    pub age_days: i64,
37    /// Files and directories to remove, across both trees.
38    pub paths: Vec<PathBuf>,
39    pub bytes: u64,
40}

42/// One desktop app session entry a purge retires with its transcript.
43#[derive(Debug, Clone, PartialEq, Eq)]
44pub struct DesktopEntry {
45    /// The entry file, on this machine or in the sync repository.
46    pub path: PathBuf,
47    /// The purged transcript it points at: what its tombstone names.
48    pub transcript_id: String,
49    /// The entry's repo-relative path when it is the repository's copy, for
50    /// the records that file it under that key.
51    pub repo_rel: Option<String>,
52}

54/// What a purge would remove, computed before anything is deleted.
55#[derive(Debug, Default)]
56pub struct PurgePlan {
57    pub retention_days: u32,
58    pub cutoff: Option<DateTime<Utc>>,
59    pub targets: Vec<PurgeTarget>,
60    /// Desktop app session entries — this machine's and the repository's —
61    /// that point at a purged transcript (see [`retire_desktop_entries`]).
62    pub desktop_entries: Vec<DesktopEntry>,
63    /// Sessions kept because at least one copy has no readable age.
64    pub undated: usize,
65    /// Transcript files that failed to parse, such as one being written right
66    /// now. Their session is kept.
67    pub unreadable: usize,
68}
69
70impl PurgePlan {
71    pub fn is_empty(&self) -> bool {
72        self.targets.is_empty()
73    }
74
75    pub fn total_bytes(&self) -> u64 {
76        self.targets.iter().map(|t| t.bytes).sum()
77    }
78
79    /// Age of the oldest session in the plan, in days.
80    pub fn oldest_age_days(&self) -> i64 {
81        self.targets.iter().map(|t| t.age_days).max().unwrap_or(0)
82    }
83}

85/// What a purge removed.
86#[derive(Debug, Default)]
87pub struct PurgeReport {
88    pub sessions: usize,
89    pub files: usize,
90    pub bytes: u64,
91    /// Desktop app session entries removed, each with a tombstone written.
92    pub desktop_entries: usize,
93    /// Paths that could not be removed, with the reason.
94    pub failures: Vec<String>,
95}

97/// Retention window in days: the longer of [`MINIMUM_RETENTION_DAYS`] and this
98/// machine's `cleanupPeriodDays`, lengthened (never shortened) by a configured
99/// override. A purge can run unattended after every sync, so a short window
100/// set once, or left in an older config, must not quietly delete recent work.
101pub fn retention_days(claude_dir: &Path, configured: Option<u32>) -> u32 {
102    let floor = MINIMUM_RETENTION_DAYS.max(claude_code_retention_days(claude_dir));
103    match configured {
104        Some(days) if days < floor => {
105            log::warn!(
106                "Purge window of {days} days is below the {floor}-day minimum; using {floor}"
107            );
108            floor
109        }
110        Some(days) => days,
111        None => floor,
112    }
113}
114
115/// `cleanupPeriodDays` from the user-level settings files, or
116/// [`CLAUDE_CODE_DEFAULT_RETENTION_DAYS`].
117///
118/// The longest of them wins rather than the one Claude Code's precedence picks:
119/// the value only ever holds a deletion back. Project-level settings are not
120/// consulted; `--older-than` covers that case.
121fn claude_code_retention_days(claude_dir: &Path) -> u32 {
122    let configured = ["settings.json", "settings.local.json"]
123        .into_iter()
124        .filter_map(|name| std::fs::read_to_string(claude_dir.join(name)).ok())
125        .filter_map(|text| serde_json::from_str::<Value>(&text).ok())
126        .filter_map(|settings| {
127            settings
128                .get("cleanupPeriodDays")
129                .and_then(Value::as_u64)
130                .filter(|days| *days > 0)
131        })
132        .map(|days| u32::try_from(days).unwrap_or(u32::MAX))
133        .max();
134
135    configured.unwrap_or(CLAUDE_CODE_DEFAULT_RETENTION_DAYS)
136}

138/// Every copy of one session found across the trees, before deciding its fate.
139#[derive(Default)]
140struct SessionCopies {
141    last_activity: Option<DateTime<Utc>>,
142    /// Set when any copy has no readable age, which keeps the session.
143    undatable: bool,
144    paths: Vec<PathBuf>,
145    bytes: u64,
146}
147
148/// Classify what a purge would remove from both project trees, without
149/// deleting anything.
150///
151/// A session is decided as a whole — every copy of it, with its sidecars — and
152/// its age is that of the most recent copy.
153pub fn plan(
154    claude_projects_dir: &Path,
155    repo_projects_dir: &Path,
156    retention_days: u32,
157) -> Result<PurgePlan> {
158    let mut plan = PurgePlan {
159        retention_days,
160        ..Default::default()
161    };
162    // Checked: a window large enough to overflow has nothing to purge anyway.
163    let Some(cutoff) = Utc::now().checked_sub_signed(Duration::days(i64::from(retention_days)))
164    else {
165        return Ok(plan);
166    };
167    plan.cutoff = Some(cutoff);
168
169    let mut by_session: std::collections::BTreeMap<String, SessionCopies> =
170        std::collections::BTreeMap::new();
171
172    for transcripts_root in [claude_projects_dir, repo_projects_dir] {
173        // Summarizing is the whole cost of planning a purge, and every file is
174        // independent: read them across the cores, then fold the results in the
175        // order the walk found them.
176        let summarized: Vec<(PathBuf, Result<crate::parser::ConversationSession>)> =
177            transcripts(transcripts_root)
178                .into_par_iter()
179                .map(|transcript| {
180                    let summary = crate::parser::ConversationSession::from_file(&transcript);
181                    (transcript, summary)
182                })
183                .collect();
184
185        for (transcript, summary) in summarized {
186            let session = match summary {
187                Ok(session) => session,
188                Err(error) => {
189                    // The session is kept whole: a copy that cannot be read
190                    // says nothing about the age of the one that can.
191                    log::warn!("Skipping unreadable {}: {error}", transcript.display());
192                    plan.unreadable += 1;
193                    if let Some(stem) = transcript.file_stem().and_then(|s| s.to_str()) {
194                        by_session.entry(stem.to_string()).or_default().undatable = true;
195                    }
196                    continue;
197                }
198            };
199
200            let paths: Vec<PathBuf> = std::iter::once(transcript.clone())
201                .chain(sidecars(&transcript))
202                .collect();
203            let bytes: u64 = paths.iter().map(|path| path_size(path)).sum();
204            let last_activity = last_activity(&session);
205
206            let copies = by_session.entry(session.session_id.clone()).or_default();
207            copies.paths.extend(paths);
208            copies.bytes += bytes;
209            copies.undatable |= last_activity.is_none();
210            copies.last_activity = copies.last_activity.max(last_activity);
211        }
212    }
213
214    let now = Utc::now();
215    for (session_id, copies) in by_session {
216        if copies.undatable {
217            plan.undated += 1;
218            continue;
219        }
220        let Some(last_activity) = copies.last_activity else {
221            continue;
222        };
223        if last_activity > cutoff {
224            continue;
225        }
226        plan.targets.push(PurgeTarget {
227            session_id,
228            last_activity,
229            age_days: (now - last_activity).num_days(),
230            paths: copies.paths,
231            bytes: copies.bytes,
232        });
233    }
234
235    plan.targets.sort_by_key(|target| target.last_activity);
236    Ok(plan)
237}

239/// Add to the plan every desktop app session entry — on this machine under
240/// `local_root`, when the app's directory is there, and the repository's —
241/// that points at a transcript the plan removes. Applying the plan then
242/// removes each and writes the tombstone beside it, the way the app retires
243/// a deleted session: the next sync drops the entry from every other
244/// machine, and none re-adopts the transcript.
245pub fn retire_desktop_entries(plan: &mut PurgePlan, local_root: Option<&Path>, repo_root: &Path) {
246    let ids: HashSet<String> = plan
247        .targets
248        .iter()
249        .map(|target| target.session_id.clone())
250        .collect();
251    let mut entries = Vec::new();
252    if let Some(root) = local_root {
253        for (path, transcript_id) in desktop_sessions::entries_for_transcripts(root, &ids) {
254            entries.push(DesktopEntry {
255                path,
256                transcript_id,
257                repo_rel: None,
258            });
259        }
260    }
261    let category_root = desktop_sessions::repo_category_root(repo_root);
262    for (path, transcript_id) in desktop_sessions::entries_for_transcripts(&category_root, &ids) {
263        let repo_rel = path
264            .strip_prefix(repo_root)
265            .ok()
266            .map(|rel| rel.to_string_lossy().replace('\\', "/"));
267        entries.push(DesktopEntry {
268            path,
269            transcript_id,
270            repo_rel,
271        });
272    }
273    plan.desktop_entries = entries;
274}

276/// Drop the per-machine record entries (last-synced base, tracked path) of
277/// the repository entries the plan retires: they describe files that exist
278/// nowhere any more, and nothing later prunes them.
279pub fn forget_desktop_records(claude_dir: &Path, repo_root: &Path, plan: &PurgePlan) -> Result<()> {
280    let removals: Vec<String> = plan
281        .desktop_entries
282        .iter()
283        .filter_map(|entry| entry.repo_rel.clone())
284        .collect();
285    if removals.is_empty() {
286        return Ok(());
287    }
288    crate::artifacts::bases::save_delta(
289        claude_dir,
290        repo_root,
291        Default::default(),
292        removals.clone(),
293    )?;
294    crate::artifacts::tracked::save_delta(claude_dir, repo_root, Default::default(), removals)
295}

297/// Delete everything the plan names. A path that is already gone is skipped,
298/// not an error.
299pub fn apply(plan: &PurgePlan) -> Result<PurgeReport> {
300    let mut report = PurgeReport::default();
301    let now_ms = Utc::now().timestamp_millis();
302
303    for entry in &plan.desktop_entries {
304        match std::fs::remove_file(&entry.path) {
305            Ok(()) => report.desktop_entries += 1,
306            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
307            Err(error) => {
308                report
309                    .failures
310                    .push(format!("{}: {error}", entry.path.display()));
311                continue;
312            }
313        }
314        // The tombstone the app would have written: epoch milliseconds, and
315        // an existing one (an earlier deletion) is left as it is.
316        let tombstone = desktop_sessions::tombstone_beside(&entry.path, &entry.transcript_id);
317        if !tombstone.exists() {
318            if let Err(error) = std::fs::write(&tombstone, now_ms.to_string()) {
319                report
320                    .failures
321                    .push(format!("{}: {error}", tombstone.display()));
322            }
323        }
324    }
325
326    for target in &plan.targets {
327        report.sessions += 1;
328        for path in &target.paths {
329            let bytes = path_size(path);
330            let removed = if path.is_dir() {
331                std::fs::remove_dir_all(path)
332            } else if path.is_file() {
333                std::fs::remove_file(path)
334            } else {
335                continue;
336            };
337            match removed {
338                Ok(()) => {
339                    report.files += 1;
340                    report.bytes += bytes;
341                }
342                Err(error) => report.failures.push(format!("{}: {error}", path.display())),
343            }
344        }
345    }
346
347    Ok(report)
348}

350/// Open the sync repository, before anything is deleted: a repository that
351/// cannot record the removal must stop the purge while the files still exist.
352pub fn open_sync_repo(repo_root: &Path) -> Result<Box<dyn crate::scm::Scm>> {
353    let repo = crate::scm::open(repo_root)
354        .with_context(|| format!("Failed to open repository at {}", repo_root.display()))?;
355    repo.has_changes()
356        .with_context(|| format!("Cannot read the repository at {}", repo_root.display()))?;
357    Ok(repo)
358}

360/// Refuse to purge while the repository holds uncommitted work.
361///
362/// A repo-side transcript that is not committed yet — a push that was declined,
363/// a path an ignore rule covers — would be deleted with no copy in history,
364/// which is the one thing a purge promises not to do.
365pub fn ensure_nothing_uncommitted(repo: &dyn crate::scm::Scm) -> Result<()> {
366    if repo.has_changes()? {
367        bail!(
368            "the sync repository has uncommitted changes; run `claude-code-sync push` \
369             first, so a purge cannot delete a copy that was never committed"
370        );
371    }
372    Ok(())
373}

375/// Stage and commit the removals. Returns whether a commit was made.
376pub fn commit_removals(repo: &dyn crate::scm::Scm, plan: &PurgePlan) -> Result<bool> {
377    repo.stage_all()?;
378    if !repo.has_changes()? {
379        return Ok(false);
380    }
381    repo.commit(&format!(
382        "Purge {} transcripts older than {} days",
383        plan.targets.len(),
384        plan.retention_days
385    ))?;
386    Ok(true)
387}

389/// Session transcripts of a project tree: `<project>/<session>.jsonl` only.
390///
391/// Subagent transcripts live deeper, under `<session>/subagents/`, and are
392/// removed with their parent session rather than on their own age.
393fn transcripts(projects_dir: &Path) -> Vec<PathBuf> {
394    if !projects_dir.is_dir() {
395        return Vec::new();
396    }
397    walkdir::WalkDir::new(projects_dir)
398        .follow_links(false)
399        .min_depth(2)
400        .max_depth(2)
401        .into_iter()
402        .filter_map(|entry| entry.ok())
403        .filter(|entry| entry.file_type().is_file())
404        .map(|entry| entry.path().to_path_buf())
405        .filter(|path| path.extension().and_then(|e| e.to_str()) == Some("jsonl"))
406        .collect()
407}

409/// Files Claude Code keeps beside a transcript and deletes with it: the
410/// session directory (`subagents/`, `tool-results/`) and any set-aside copies.
411fn sidecars(transcript: &Path) -> Vec<PathBuf> {
412    let Some(directory) = transcript.parent() else {
413        return Vec::new();
414    };
415    let Some(stem) = transcript.file_stem().and_then(|s| s.to_str()) else {
416        return Vec::new();
417    };
418
419    let mut found = Vec::new();
420    let session_directory = directory.join(stem);
421    if session_directory.is_dir() {
422        found.push(session_directory);
423    }
424
425    // Every sibling named after this session: `.orphaned-…`, `.superseded-…`,
426    // `.stopoffset`, and whatever Claude Code adds next.
427    let companion = format!("{stem}.");
428    let transcript_name = transcript.file_name();
429    let Ok(entries) = std::fs::read_dir(directory) else {
430        return found;
431    };
432    for entry in entries.filter_map(|entry| entry.ok()) {
433        if Some(entry.file_name().as_os_str()) == transcript_name {
434            continue;
435        }
436        let Some(name) = entry.file_name().to_str().map(str::to_string) else {
437            continue;
438        };
439        if name.starts_with(&companion) {
440            found.push(entry.path());
441        }
442    }
443    found
444}

446/// Timestamp of the session's last message, in UTC.
447fn last_activity(session: &crate::parser::ConversationSession) -> Option<DateTime<Utc>> {
448    let stamp = session.latest_timestamp()?;
449    DateTime::parse_from_rfc3339(stamp)
450        .ok()
451        .map(|dated| dated.with_timezone(&Utc))
452}
453
454fn path_size(path: &Path) -> u64 {
455    if path.is_dir() {
456        return walkdir::WalkDir::new(path)
457            .follow_links(false)
458            .into_iter()
459            .filter_map(|entry| entry.ok())
460            .filter_map(|entry| entry.metadata().ok())
461            .filter(|metadata| metadata.is_file())
462            .map(|metadata| metadata.len())
463            .sum();
464    }
465    std::fs::metadata(path).map(|m| m.len()).unwrap_or(0)
466}
