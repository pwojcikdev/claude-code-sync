1//! The category registry: one row per syncable artifact kind. Everything the
2//! engine does — collecting sources, choosing repo destinations, picking a
3//! merge strategy — is driven by this data.
4
5use serde::{Deserialize, Serialize};
6
7/// Identifies one syncable artifact category.
8#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
9#[serde(rename_all = "kebab-case")]
10pub enum CategoryId {
11    Settings,
12    Memory,
13    Skills,
14    Agents,
15    Commands,
16    Rules,
17    Hooks,
18    Plugins,
19    Plans,
20    Todos,
21    PromptHistory,
22    /// Non-JSONL files inside `~/.claude/projects/` (images, PDFs, per-project
23    /// memory). Gated by `!exclude_attachments` instead of a toggle, honoring
24    /// the long-documented attachments contract.
25    ProjectAttachments,
26    /// The Claude desktop app's session list (its Code tab sidebar): one
27    /// entry per session plus deletion tombstones, kept outside `~/.claude`
28    /// in the app's own data directory.
29    DesktopSessions,
30}
31
32/// What to copy, relative to `~/.claude`. Missing sources are treated as
33/// empty, never as an error — a machine may simply not have e.g. `todos/`.
34#[derive(Debug, Clone, Copy, PartialEq, Eq)]
35pub enum SourceSpec {
36    /// Exact files. Anything not listed is structurally unreachable — this is
37    /// how `plugins/` syncs its two manifests but never `cache/`.
38    Files(&'static [&'static str]),
39    /// A directory walked recursively (symlinks not followed).
40    Dir(&'static str),
41    /// The desktop app's `claude-code-sessions/` directory, outside
42    /// `~/.claude` (see `super::desktop_sessions::dir`): only its
43    /// `<org>/<account>/local_*.json` entries and `deleted_*` tombstones.
44    DesktopSessions,
45}
46
47/// How pull reconciles a repo copy with the local file.
48#[derive(Debug, Clone, Copy, PartialEq, Eq)]
49pub enum MergeStrategy {
50    /// Remote bytes win (only written when they differ; snapshot first).
51    RawOverwrite,
52    /// Line-wise union of both sides (append-only JSONL like history.jsonl);
53    /// applied on push and pull so machines converge grow-only.
54    UnionJsonl,
55    /// Union of `MEMORY.md` indexes by link target, applied on push and pull;
56    /// every other file in the category raw-overwrites. Overwriting an index
57    /// with a sparser one orphans the other machine's memory files.
58    UnionMemoryIndex,
59    /// Desktop session entries: remote wins when only the other machine
60    /// changed one, and a file both machines changed is merged field by
61    /// field (see `super::desktop_sessions`) instead of held back — the
62    /// app rewrites every entry it opens, so holding back would never
63    /// settle. Tombstones keep the later deletion time.
64    SessionIndex,
65}
66
67/// Where a category's files live inside the sync repository.
68#[derive(Debug, Clone, Copy, PartialEq, Eq)]
69pub enum DestRoot {
70    /// `<repo>/artifacts/<repo_subdir>/...` — the default for all categories.
71    Artifacts,
72    /// `<repo>/<sync_subdirectory>/...` — shares the session tree (used only
73    /// by project attachments, which live beside their transcripts).
74    SessionTree,
75}
76
77/// One registry row.
78#[derive(Debug)]
79pub struct CategoryDescriptor {
80    pub id: CategoryId,
81    /// Config key / CLI name (kebab-case).
82    pub name: &'static str,
83    /// Subdirectory under the sync repo's `artifacts/` root (unused for
84    /// `DestRoot::SessionTree`).
85    pub repo_subdir: &'static str,
86    pub source: SourceSpec,
87    pub merge: MergeStrategy,
88    pub dest: DestRoot,
89    /// File extensions this category never copies (e.g. attachments skip
90    /// `.jsonl`, which belongs to the session pipeline).
91    pub exclude_extensions: &'static [&'static str],
92    /// Rewrite this machine's absolute paths to tokens into the repo and back
93    /// out. For config files that embed paths; user content is stored verbatim.
94    pub tokenize_paths: bool,
95    /// Also tokenize mapped projects (`__PROJECT_<id>__`), so a path into a
96    /// project resolves to wherever each machine keeps it. A file naming a
97    /// project this machine has not mapped is skipped on pull, like the
98    /// project's transcripts are.
99    pub tokenize_projects: bool,
100    /// Propagate deletions: a file this machine synced before and no longer has
101    /// is removed from the other side. Only for curated directories.
102    pub mirror_deletes: bool,
103    /// Short human description for wizard / config --show.
104    pub description: &'static str,
105}

108/// Deliberately a constant: a fixed category→path mapping keeps pull
109/// deterministic and the deny-guarantee auditable.

287/// Per-category enable switches, stored in the `[sync_artifacts]` table of
288/// config.toml. Every field defaults to false so configs written by older
289/// versions parse unchanged and existing users opt in explicitly.
290#[derive(Debug, Clone, Default, Serialize, Deserialize, PartialEq, Eq)]
291pub struct ArtifactToggles {
292    #[serde(default)]
293    pub settings: bool,
294    #[serde(default)]
295    pub memory: bool,
296    #[serde(default)]
297    pub skills: bool,
298    #[serde(default)]
299    pub agents: bool,
300    #[serde(default)]
301    pub commands: bool,
302    #[serde(default)]
303    pub rules: bool,
304    #[serde(default)]
305    pub hooks: bool,
306    #[serde(default)]
307    pub plugins: bool,
308    #[serde(default)]
309    pub plans: bool,
310    #[serde(default)]
311    pub todos: bool,
312    #[serde(default)]
313    pub prompt_history: bool,
314    #[serde(default)]
315    pub desktop_sessions: bool,
316}
317
318impl ArtifactToggles {
319    /// A toggles value with every category enabled, `hooks` included.
320    pub fn all_enabled() -> Self {
321        Self {
322            settings: true,
323            memory: true,
324            skills: true,
325            agents: true,
326            commands: true,
327            rules: true,
328            hooks: true,
329            plugins: true,
330            plans: true,
331            todos: true,
332            prompt_history: true,
333            desktop_sessions: true,
334        }
335    }
336
337    /// Every category except `hooks` (used by onboarding defaults and the
338    /// `all` CLI shorthand). Hooks run whatever the sync repository hands
339    /// them, so they are only ever switched on by name.
340    pub fn all_but_hooks() -> Self {
341        Self {
342            hooks: false,
343            ..Self::all_enabled()
344        }
345    }
346
347    /// Whether any category is enabled at all.
348    pub fn any_enabled(&self) -> bool {
349        REGISTRY.iter().any(|d| self.is_enabled(d.id))
350    }
351
352    /// Read the switch for one category.
353    pub fn is_enabled(&self, id: CategoryId) -> bool {
354        match id {
355            CategoryId::Settings => self.settings,
356            CategoryId::Memory => self.memory,
357            CategoryId::Skills => self.skills,
358            CategoryId::Agents => self.agents,
359            CategoryId::Commands => self.commands,
360            CategoryId::Rules => self.rules,
361            CategoryId::Hooks => self.hooks,
362            CategoryId::Plugins => self.plugins,
363            CategoryId::Plans => self.plans,
364            CategoryId::Todos => self.todos,
365            CategoryId::PromptHistory => self.prompt_history,
366            CategoryId::DesktopSessions => self.desktop_sessions,
367            // Attachments are gated by FilterConfig::exclude_attachments, not
368            // by a toggle; as a toggle they always read as off.
369            CategoryId::ProjectAttachments => false,
370        }
371    }
372
373    /// Flip the switch for one category.
374    pub fn set_enabled(&mut self, id: CategoryId, value: bool) {
375        match id {
376            CategoryId::Settings => self.settings = value,
377            CategoryId::Memory => self.memory = value,
378            CategoryId::Skills => self.skills = value,
379            CategoryId::Agents => self.agents = value,
380            CategoryId::Commands => self.commands = value,
381            CategoryId::Rules => self.rules = value,
382            CategoryId::Hooks => self.hooks = value,
383            CategoryId::Plugins => self.plugins = value,
384            CategoryId::Plans => self.plans = value,
385            CategoryId::Todos => self.todos = value,
386            CategoryId::PromptHistory => self.prompt_history = value,
387            CategoryId::DesktopSessions => self.desktop_sessions = value,
388            CategoryId::ProjectAttachments => {}
389        }
390    }
391}

393/// Look up a descriptor by its CLI/config name (kebab-case).
394pub fn find_by_name(name: &str) -> Option<&'static CategoryDescriptor> {
395    REGISTRY.iter().find(|d| d.name == name)
396}

398/// Registry rows driven by ArtifactToggles — everything except attachments,
399/// which the long-standing exclude_attachments flag governs.
400pub fn toggleable() -> impl Iterator<Item = &'static CategoryDescriptor> {
401    REGISTRY
402        .iter()
403        .filter(|d| d.id != CategoryId::ProjectAttachments)
404}
405
406/// All registry rows whose toggle is on.
407#[allow(dead_code)] // used via the library target; the bin compiles this module separately
408pub fn enabled_categories(toggles: &ArtifactToggles) -> Vec<&'static CategoryDescriptor> {
409    REGISTRY
410        .iter()
411        .filter(|d| toggles.is_enabled(d.id))
412        .collect()
413}

455    #[test]
456    fn test_prompt_history_uses_union_merge() {
457        let ph = REGISTRY
458            .iter()
459            .find(|d| d.id == CategoryId::PromptHistory)
460            .unwrap();
461        assert_eq!(ph.merge, MergeStrategy::UnionJsonl);
462        assert_eq!(ph.source, SourceSpec::Files(&["history.jsonl"]));
463        // Only project attachments, which carry per-project MEMORY.md indexes,
464        // and the desktop session list merge as well; everything else
465        // raw-overwrites.
466        for d in REGISTRY.iter().filter(|d| {
467            d.id != CategoryId::PromptHistory
468                && d.id != CategoryId::ProjectAttachments
469                && d.id != CategoryId::DesktopSessions
470        }) {
471            assert_eq!(d.merge, MergeStrategy::RawOverwrite, "{}", d.name);
472        }
473        let attachments = REGISTRY
474            .iter()
475            .find(|d| d.id == CategoryId::ProjectAttachments)
476            .unwrap();
477        assert_eq!(attachments.merge, MergeStrategy::UnionMemoryIndex);
478        let desktop = REGISTRY
479            .iter()
480            .find(|d| d.id == CategoryId::DesktopSessions)
481            .unwrap();
482        assert_eq!(desktop.merge, MergeStrategy::SessionIndex);
483        assert_eq!(desktop.source, SourceSpec::DesktopSessions);
484        assert!(
485            !desktop.mirror_deletes,
486            "deletions travel as the app's own tombstones, not through the tracked record"
487        );
488    }

559    #[test]
560    fn test_toggles_map_every_category() {
561        let mut toggles = ArtifactToggles::default();
562        assert!(!toggles.any_enabled());
563        for d in toggleable() {
564            assert!(!toggles.is_enabled(d.id), "{} defaults off", d.name);
565            toggles.set_enabled(d.id, true);
566            assert!(toggles.is_enabled(d.id), "{} can be enabled", d.name);
567        }
568        assert_eq!(toggles, ArtifactToggles::all_enabled());
569        // Attachments are flag-governed, not toggleable.
570        assert!(!toggles.is_enabled(CategoryId::ProjectAttachments));
571        toggles.set_enabled(CategoryId::ProjectAttachments, true);
572        assert!(!toggles.is_enabled(CategoryId::ProjectAttachments));
573    }
